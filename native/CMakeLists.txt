cmake_minimum_required(VERSION 3.22)
project(golos_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# Baseline x64 build also runs on CPUs without AVX/AVX2. ARM64 uses NEON.
foreach(flag AVX AVX2 FMA F16C AVX_VNNI AVX512 AVX512_VBMI AVX512_VNNI)
  set(GGML_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT DEFINED WHISPER_SOURCE)
  message(FATAL_ERROR "Set WHISPER_SOURCE to whisper.cpp v1.8.2")
endif()
add_subdirectory(${WHISPER_SOURCE} whisper)
add_library(golos_native SHARED WhisperBridge.cpp)
target_include_directories(golos_native PRIVATE ${WHISPER_SOURCE}/include ${WHISPER_SOURCE}/ggml/include)
target_link_libraries(golos_native PRIVATE whisper)
